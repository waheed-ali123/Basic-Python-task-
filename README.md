hours = float(input("Enter hours worked: "))

fahrenheit = (celsius * 9/5) + 32
print("Temperature in Fahrenheit:", fahrenheit)

num2 = float(input("Enter the second number: "))
result = num1 ** num2
print(f"{num1} raised to the power of {num2} is {result}")
